import { Component, inject, input, OnInit, signal } from '@angular/core';
import { AbstractControl, FormBuilder, ReactiveFormsModule, ValidationErrors, Validators } from '@angular/forms';
import { Router, RouterLink } from '@angular/router';
import { forkJoin } from 'rxjs';
import { idDe } from '../../../core/models/referencia';
import { PetsService } from '../../pets/services/pets.service';
import { ServicosService } from '../../servicos/services/servicos.service';
import { ProfissionaisService } from '../../profissionais/services/profissionais.service';
import { Pet } from '../../pets/pet.model';
import { Servico } from '../../servicos/servico.model';
import { Profissional } from '../../profissionais/profissional.model';
import { AtendimentosService } from '../services/atendimentos.service';
function agendamentoFuturo(campo: AbstractControl): ValidationErrors | null {
  const valor = String(campo.value ?? '');
  if (!valor) return null;
  const data = new Date(valor);
  return Number.isNaN(data.getTime()) || data <= new Date() ? { dataInvalida: true } : null;
}

@Component({
  selector: 'app-atendimento-form',
  imports: [ReactiveFormsModule, RouterLink],
  templateUrl: './atendimento-form.html',
})
export class AtendimentoForm implements OnInit {
  private readonly fb = inject(FormBuilder);
  private readonly api = inject(AtendimentosService);
  private readonly petsService = inject(PetsService);
  private readonly servicosService = inject(ServicosService);
  private readonly profissionaisService = inject(ProfissionaisService);
  private readonly router = inject(Router);
  readonly id = input<string>();
  protected readonly pets = signal<Pet[]>([]);
  protected readonly servicos = signal<Servico[]>([]);
  protected readonly profissionais = signal<Profissional[]>([]);
  protected readonly carregando = signal(true);
  protected readonly salvando = signal(false);
  protected readonly erro = signal('');
  protected readonly form = this.fb.nonNullable.group({
    pet: ['', Validators.required],
    servico: ['', Validators.required],
    profissional: ['', Validators.required],
    dataHora: ['', [Validators.required, agendamentoFuturo]],
    observacoes: ['', Validators.maxLength(300)],
  });
  ngOnInit(): void {
    forkJoin({
      pets: this.petsService.listar(),
      servicos: this.servicosService.listar(),
      profissionais: this.profissionaisService.listar(),
    }).subscribe({
      next: ({ pets, servicos, profissionais }) => {
        this.pets.set(pets);
        this.servicos.set(servicos);
        this.profissionais.set(profissionais.filter((x) => x.ativo));
        const id = this.id();
        if (!id) {
          this.carregando.set(false);
          return;
        }
        this.api.buscarPorId(id).subscribe({
          next: (a) => {
            this.form.patchValue({
              pet: idDe(a.pet),
              servico: idDe(a.servico),
              profissional: idDe(a.profissional),
              dataHora: this.paraLocal(a.dataHora),
              observacoes: a.observacoes ?? '',
            });
            this.carregando.set(false);
          },
          error: (e: Error) => {
            this.erro.set(e.message);
            this.carregando.set(false);
          },
        });
      },
      error: (e: Error) => {
        this.erro.set(e.message);
        this.carregando.set(false);
      },
    });
  }
  protected invalido(campo: keyof typeof this.form.controls): boolean {
    const c = this.form.controls[campo];
    return c.invalid && (c.dirty || c.touched);
  }
  protected salvar(): void {
    if (this.form.invalid) {
      this.form.markAllAsTouched();
      return;
    }
    const dados = this.form.getRawValue();
    if (
      Number.isNaN(new Date(dados.dataHora).getTime()) ||
      new Date(dados.dataHora) <= new Date()
    ) {
      this.erro.set('Selecione data e hora futuras.');
      return;
    }
    this.salvando.set(true);
    this.erro.set('');
    const req = this.id() ? this.api.atualizar(this.id()!, dados) : this.api.criar(dados);
    req.subscribe({
      next: () => void this.router.navigate(['/atendimentos']),
      error: (e: Error) => {
        this.erro.set(e.message);
        this.salvando.set(false);
      },
    });
  }
  private paraLocal(iso: string): string {
    const data = new Date(iso);
    if (Number.isNaN(data.getTime())) return iso.slice(0, 16);
    return new Date(data.getTime() - data.getTimezoneOffset() * 60000).toISOString().slice(0, 16);
  }
}

import { DatePipe } from '@angular/common';
import { Component, inject, OnInit, signal } from '@angular/core';
import { RouterLink } from '@angular/router';
import { forkJoin } from 'rxjs';
import { idDe, Referencia } from '../../../core/models/referencia';
import { ConfirmacaoService } from '../../../core/ui/confirmacao.service';
import { PetsService } from '../../pets/services/pets.service';
import { ServicosService } from '../../servicos/services/servicos.service';
import { ProfissionaisService } from '../../profissionais/services/profissionais.service';
import { Atendimento, STATUS, Status, TRANSICOES } from '../atendimento.model';
import { AtendimentosService } from '../services/atendimentos.service';
@Component({
  selector: 'app-atendimentos-lista',
  imports: [DatePipe, RouterLink],
  templateUrl: './atendimentos-lista.html',
})
export class AtendimentosLista implements OnInit {
  private readonly api = inject(AtendimentosService);
  private readonly confirmacao = inject(ConfirmacaoService);
  private readonly petsService = inject(PetsService);
  private readonly servicosService = inject(ServicosService);
  private readonly profissionaisService = inject(ProfissionaisService);
  protected readonly estados = STATUS;
  protected readonly transicoes = TRANSICOES;
  protected readonly itens = signal<Atendimento[]>([]);
  protected readonly carregando = signal(true);
  protected readonly operando = signal(false);
  protected readonly erro = signal('');
  protected readonly sucesso = signal('');
  protected readonly status = signal<Status | ''>('');
  protected readonly nomesPets = signal<Record<string, string>>({});
  protected readonly nomesServicos = signal<Record<string, string>>({});
  protected readonly nomesProfissionais = signal<Record<string, string>>({});
  ngOnInit(): void {
    forkJoin({
      pets: this.petsService.listar(),
      servicos: this.servicosService.listarTodos(),
      profissionais: this.profissionaisService.listar(),
    }).subscribe({
      next: ({ pets, servicos, profissionais }) => {
        this.nomesPets.set(Object.fromEntries(pets.map((x) => [x._id, x.nome])));
        this.nomesServicos.set(Object.fromEntries(servicos.map((x) => [x._id, x.nome])));
        this.nomesProfissionais.set(Object.fromEntries(profissionais.map((x) => [x._id, x.nome])));
      },
      error: () => this.erro.set('Não foi possível carregar alguns nomes de referência.'),
    });
    this.carregar();
  }
  protected nome(ref: Referencia, tipo: 'pet' | 'servico' | 'profissional'): string {
    const id = idDe(ref);
    const mapa =
      tipo === 'pet'
        ? this.nomesPets()
        : tipo === 'servico'
          ? this.nomesServicos()
          : this.nomesProfissionais();
    return (
      mapa[id] ??
      (typeof ref === 'object' && typeof ref['nome'] === 'string' ? String(ref['nome']) : id)
    );
  }
  protected filtrar(valor: string): void {
    this.status.set(valor as Status | '');
    this.carregar();
  }
  protected carregar(): void {
    this.carregando.set(true);
    this.erro.set('');
    this.api.listar(this.status() || undefined).subscribe({
      next: (xs) => {
        this.itens.set(xs);
        this.carregando.set(false);
      },
      error: (e: Error) => {
        this.erro.set(e.message);
        this.carregando.set(false);
      },
    });
  }
  protected async avancar(a: Atendimento, destino: Status): Promise<void> {
    if (this.operando()) return;
    const cancelamento = destino === 'cancelado';
    const verbo = destino === 'em_atendimento' ? 'Iniciar' : destino === 'concluido' ? 'Concluir' : 'Cancelar';
    const aceitou = await this.confirmacao.confirmar({
      titulo: cancelamento ? 'Cancelar atendimento?' : `${verbo} atendimento?`,
      descricao: cancelamento
        ? `Tem certeza que deseja cancelar o atendimento de ${this.nome(a.pet, 'pet')}?`
        : `Confirma a mudança de status do atendimento de ${this.nome(a.pet, 'pet')} para "${destino.replace('_', ' ')}"?`,
      textoConfirmar: cancelamento ? 'Sim, cancelar' : `Sim, ${verbo.toLowerCase()}`,
      tipo: cancelamento ? 'perigo' : 'padrao',
    });
    if (!aceitou) return;
    this.operando.set(true);
    this.erro.set('');
    this.sucesso.set('');
    this.api.mudarStatus(a._id, destino).subscribe({
      next: () => {
        this.operando.set(false);
        this.sucesso.set('Status alterado com sucesso.');
        this.carregar();
      },
      error: (e: Error) => {
        this.operando.set(false);
        this.erro.set(e.message);
      },
    });
  }
}

import { delay, Observable, of, throwError } from 'rxjs';
import { idDe } from '../../../core/models/referencia';
import { Atendimento, AtendimentoEntrada, Status, TRANSICOES } from '../atendimento.model';
import { AtendimentosService } from './atendimentos.service';

/** Somente demonstração C1: segurança efetiva das regras exige validação no back-end. */
@Injectable()
export class AtendimentosMockService extends AtendimentosService {
  private readonly atraso = 300;
  private proximoId = 4;
  private readonly futuro = (dias: number, hora: number) => {
    const data = new Date();
    data.setDate(data.getDate() + dias);
    data.setHours(hora, 0, 0, 0);
    const local = new Date(data.getTime() - data.getTimezoneOffset() * 60000);
    return local.toISOString().slice(0, 16);
  };
  private atendimentos: Atendimento[] = [
    {
      _id: '1',
      pet: '1',
      servico: '1',
      profissional: '1',
      dataHora: this.futuro(3, 10),
      status: 'agendado',
      observacoes: 'Pet com medo de secador',
    },
    {
      _id: '2',
      pet: '2',
      servico: '2',
      profissional: '2',
      dataHora: this.futuro(4, 14),
      status: 'agendado',
    },
    {
      _id: '3',
      pet: '3',
      servico: '1',
      profissional: '1',
      dataHora: this.futuro(5, 11),
      status: 'agendado',
    },
  ];

  listar(status?: Status): Observable<Atendimento[]> {
    return of(
      this.atendimentos
        .filter((a) => !status || a.status === status)
        .map((a) => ({ ...a }))
        .sort((a, b) => a.dataHora.localeCompare(b.dataHora)),
    ).pipe(delay(this.atraso));
  }
  buscarPorId(id: string): Observable<Atendimento> {
    const a = this.atendimentos.find((item) => item._id === id);
    return a
      ? of({ ...a }).pipe(delay(this.atraso))
      : throwError(() => new Error('Atendimento não encontrado.'));
  }
  criar(dados: AtendimentoEntrada): Observable<Atendimento> {
    const erro = this.validar(dados);
    if (erro) return throwError(() => new Error(erro));
    const novo: Atendimento = { ...dados, _id: String(this.proximoId++), status: 'agendado' };
    this.atendimentos.push(novo);
    return of({ ...novo }).pipe(delay(this.atraso));
  }
  atualizar(id: string, dados: AtendimentoEntrada): Observable<Atendimento> {
    const pos = this.atendimentos.findIndex((a) => a._id === id);
    if (pos < 0) return throwError(() => new Error('Atendimento não encontrado.'));
    if (this.atendimentos[pos].status !== 'agendado') {
      return throwError(() => new Error('Somente atendimentos agendados podem ser editados.'));
    }
    const erro = this.validar(dados, id);
    if (erro) return throwError(() => new Error(erro));
    const atualizado = { ...this.atendimentos[pos], ...dados };
    this.atendimentos[pos] = atualizado;
    return of({ ...atualizado }).pipe(delay(this.atraso));
  }
  mudarStatus(id: string, status: Status): Observable<Atendimento> {
    const a = this.atendimentos.find((item) => item._id === id);
    if (!a) return throwError(() => new Error('Atendimento não encontrado.'));
    if (!TRANSICOES[a.status].includes(status)) {
      return throwError(
        () => new Error(`Não é possível passar de "${a.status}" para "${status}".`),
      );
    }
    a.status = status;
    return of({ ...a }).pipe(delay(this.atraso));
  }
  private validar(dados: AtendimentoEntrada, idAtual?: string): string | null {
    if (!dados.pet || !dados.servico || !dados.profissional || !dados.dataHora)
      return 'Preencha todos os campos obrigatórios.';
    if (
      Number.isNaN(new Date(dados.dataHora).getTime()) ||
      new Date(dados.dataHora) <= new Date()
    ) {
      return 'O atendimento precisa ser agendado para o futuro.';
    }
    if ((dados.observacoes ?? '').length > 300)
      return 'Observações devem ter no máximo 300 caracteres.';
    const conflito = this.atendimentos.some(
      (a) =>
        a._id !== idAtual &&
        a.status !== 'cancelado' &&
        idDe(a.profissional) === dados.profissional &&
        new Date(a.dataHora).getTime() === new Date(dados.dataHora).getTime(),
    );
    if (conflito) return 'O profissional já tem um atendimento neste horário.';
    return null;
  }
}

function dataNascimentoNaoFutura(campo: AbstractControl): ValidationErrors | null {
  const valor = String(campo.value ?? '');
  if (!valor) return null;
  const agora = new Date();
  const hoje = `${agora.getFullYear()}-${String(agora.getMonth() + 1).padStart(2, '0')}-${String(agora.getDate()).padStart(2, '0')}`;
  return valor > hoje ? { dataFutura: true } : null;
}

@Component({
  selector: 'app-pet-form',
  standalone: true,
  imports: [ReactiveFormsModule],

  styleUrl: './pet-form.css',

export class PetForm {

  private readonly petsService = inject(PetsService);
  private readonly tutoresService = inject(TutoresService);
  readonly tutores = signal<Tutor[]>([]);

  readonly especies = ESPECIES;
  readonly petInicial = input<Pet | null>(null);
  readonly salvo = output<void>();
  readonly cancelar = output<void>();

  readonly salvando = signal(false);
  readonly erro = signal('');

  readonly formulario = this.fb.nonNullable.group({
    nome: ['', [Validators.required, Validators.minLength(2), Validators.maxLength(60)]],
    especie: ['cachorro', Validators.required],
    raca: ['', Validators.maxLength(60)],
    dataNascimento: ['', [dataNascimentoNaoFutura]],
    tutor: ['', Validators.required],
    ativo: [true],
  });

  constructor() {
    this.tutoresService.listar().subscribe({
      next: (tutores) => this.tutores.set(tutores),
      error: () => this.erro.set('Não foi possível carregar os tutores.'),
    });
    effect(() => {
      const pet = this.petInicial();

      if (pet) {
        this.formulario.patchValue({
          nome: pet.nome,
          especie: pet.especie,
          raca: pet.raca ?? '',
          dataNascimento: pet.dataNascimento ?? '',
          tutor: pet.tutor,
          ativo: pet.ativo,
        });
      } else {
        this.formulario.reset({
          nome: '',
          especie: 'cachorro',
          raca: '',
          dataNascimento: '',
          tutor: '',
          ativo: true,
        });
      }

      this.erro.set('');
    });
  }

  dataFutura(): boolean {
    const data = this.formulario.controls.dataNascimento.value;

    return !!data && data > new Date().toISOString().slice(0, 10);
  }

  salvar(): void {
    if (this.formulario.invalid || this.dataFutura()) {
      this.formulario.markAllAsTouched();
      return;
    }

    this.salvando.set(true);
    this.erro.set('');

    const dados = this.formulario.getRawValue();
    const pet = this.petInicial();

    const entrada = {
      ...dados,
      especie: dados.especie as (typeof ESPECIES)[number],
      raca: dados.raca || undefined,
      dataNascimento: dados.dataNascimento || undefined,
    };

    const operacao = pet
      ? this.petsService.atualizar(pet._id, entrada)
      : this.petsService.criar(entrada);

    operacao.subscribe({
      next: () => {
        this.salvando.set(false);
        this.salvo.emit();
      },
      error: (erro: Error) => {
        this.salvando.set(false);
        this.erro.set(erro.message || 'Não foi possível salvar o pet.');
      },
    });
  }

      ? this.pets.filter((pet) => pet.especie === especie)
      : this.pets;

    return of(
      [...lista].sort((a, b) => a.nome.localeCompare(b.nome)),
    ).pipe(delay(this.atraso));
  }

  buscarPorId(id: string): Observable<Pet> {
    const pet = this.pets.find((item) => item._id === id);

    return pet
      ? of({ ...pet }).pipe(delay(this.atraso))
      : throwError(() => new Error(`Pet ${id} não encontrado`));
  }

  criar(dados: PetEntrada): Observable<Pet> {
    const novo: Pet = {
      ...dados,
      _id: String(this.proximoId++),
    };

    this.pets.push(novo);

    return of({ ...novo }).pipe(delay(this.atraso));
  }

  atualizar(
    id: string,
    dados: Partial<PetEntrada>,
  ): Observable<Pet> {
    const indice = this.pets.findIndex((pet) => pet._id === id);

    if (indice < 0) {
      return throwError(() => new Error(`Pet ${id} não encontrado`));
    }

    this.pets[indice] = {
      ...this.pets[indice],
      ...dados,
      _id: id,
    };

    return of({ ...this.pets[indice] }).pipe(delay(this.atraso));
  }

  remover(id: string): Observable<void> {
    const existe = this.pets.some((pet) => pet._id === id);

    if (!existe) {
      return throwError(() => new Error(`Pet ${id} não encontrado`));
    }

    this.pets = this.pets.filter((pet) => pet._id !== id);

    return of(undefined).pipe(delay(this.atraso));

import { HttpClient, HttpErrorResponse, HttpParams } from '@angular/common/http';
import { inject, Injectable } from '@angular/core';
import { catchError, map, Observable, throwError } from 'rxjs';
import { environment } from '../../../../environments/environment';
import { RespostaApi } from '../../../core/models/resposta-api';
import { Categoria, Produto, ProdutoEntrada } from '../produto.model';
import { ProdutosService } from './produtos.service';

/**
 * Serviço que conversa com a API NestJS de verdade (PDF 02, cap. 9).
 * Ativado quando environment.usarMock = false.
 */
@Injectable()
export class ProdutosApiService extends ProdutosService {
  private readonly http = inject(HttpClient);
  private readonly url = `${environment.apiUrl}/produtos`;

  listar(categoria?: Categoria): Observable<Produto[]> {
    const params = categoria ? new HttpParams().set('categoria', categoria) : undefined;
    return this.http.get<RespostaApi<Produto[]>>(this.url, { params }).pipe(
      map((r) => r.dados),
      catchError(this.tratarErro),
    );
  }

  buscarPorId(id: string): Observable<Produto> {
    return this.http.get<RespostaApi<Produto>>(`${this.url}/${id}`).pipe(
      map((r) => r.dados),
      catchError(this.tratarErro),
    );
  }

  criar(dados: ProdutoEntrada): Observable<Produto> {
    return this.http.post<RespostaApi<Produto>>(this.url, dados).pipe(
      map((r) => r.dados),
      catchError(this.tratarErro),
    );
  }

  atualizar(id: string, dados: Partial<ProdutoEntrada>): Observable<Produto> {
    return this.http.patch<RespostaApi<Produto>>(`${this.url}/${id}`, dados).pipe(
      map((r) => r.dados),
      catchError(this.tratarErro),
    );
  }

  remover(id: string): Observable<void> {
    return this.http.delete<void>(`${this.url}/${id}`).pipe(catchError(this.tratarErro));
  }

  /** Converte o erro HTTP em uma mensagem amigável para a tela. */
  private tratarErro(erro: HttpErrorResponse): Observable<never> {
    let mensagem = 'Não foi possível falar com o servidor. Tente novamente.';
    const corpo = erro.error as { message?: string | string[] } | null;
    if (erro.status > 0 && corpo?.message) {
      mensagem = Array.isArray(corpo.message) ? corpo.message.join(' · ') : corpo.message;
    }
    return throwError(() => new Error(mensagem));
  }
}

    const lista = categoria ? this.produtos.filter((p) => p.categoria === categoria) : this.produtos;
    return of([...lista].sort((a, b) => a.nome.localeCompare(b.nome))).pipe(delay(this.atraso));
  }

  buscarPorId(id: string): Observable<Produto> {
    const produto = this.produtos.find((p) => p._id === id);
    return produto
      ? of({ ...produto }).pipe(delay(this.atraso))
      : throwError(() => new Error(`Produto ${id} não encontrado`));
  }

  criar(dados: ProdutoEntrada): Observable<Produto> {
    if (this.produtos.some((p) => p.nome === dados.nome)) {
      return throwError(() => new Error(`Já existe um produto com o nome "${dados.nome}"`));
    }
    const novo: Produto = { ...dados, _id: String(this.proximoId++), ativo: dados.ativo ?? true };
    this.produtos.push(novo);
    return of(novo).pipe(delay(this.atraso));
  }

  atualizar(id: string, dados: Partial<ProdutoEntrada>): Observable<Produto> {
    const indice = this.produtos.findIndex((p) => p._id === id);
    if (indice < 0) {
      return throwError(() => new Error(`Produto ${id} não encontrado`));
    }
    this.produtos[indice] = { ...this.produtos[indice], ...dados };
    return of(this.produtos[indice]).pipe(delay(this.atraso));
  }

  remover(id: string): Observable<void> {
    const produto = this.produtos.find((p) => p._id === id);
    if (produto && produto.estoque > 0) {
      return throwError(
        () => new Error('Não é possível excluir um produto com estoque. Zere o estoque ou inative o produto.'),
      );
    }
    this.produtos = this.produtos.filter((p) => p._id !== id);

      telefone: '(51) 99999-2222',
      email: 'maria@email.com',
      endereco: 'Rua Central, 200',
      ativo: true,
    },
    {
      _id: '3',
      nome: 'Pedro Santos',
      telefone: '(51) 99999-3333',
      email: 'pedro@email.com',
      ativo: true,
    },
  ];

  private proximoId = 4;
  private readonly atraso = 400;

  listar(): Observable<Tutor[]> {
    return of([...this.tutores]).pipe(delay(this.atraso));
  }

  buscarPorId(id: string): Observable<Tutor> {
    const tutor = this.tutores.find((t) => t._id === id);

    return tutor
      ? of({ ...tutor }).pipe(delay(this.atraso))
      : throwError(() => new Error(`Tutor ${id} não encontrado`));
  }

  criar(dados: TutorEntrada): Observable<Tutor> {
    const emailExiste = this.tutores.some(
      (t) => t.email.toLowerCase() === dados.email.toLowerCase(),
    );

    if (emailExiste) {
      return throwError(() => new Error('Já existe um tutor com esse e-mail'));
    }

    const novo: Tutor = {
      ...dados,
      _id: String(this.proximoId++),
      ativo: dados.ativo ?? true,
    };

    this.tutores.push(novo);

    return of({ ...novo }).pipe(delay(this.atraso));
  }

  atualizar(
    id: string,
    dados: Partial<TutorEntrada>,
  ): Observable<Tutor> {
    const indice = this.tutores.findIndex((t) => t._id === id);

    if (indice < 0) {
      return throwError(() => new Error(`Tutor ${id} não encontrado`));
    }

    if (
      dados.email &&
      this.tutores.some(
        (t) =>
          t._id !== id &&
          t.email.toLowerCase() === dados.email!.toLowerCase(),
      )
    ) {
      return throwError(() => new Error('Já existe um tutor com esse e-mail'));
    }

    this.tutores[indice] = { ...this.tutores[indice], ...dados };

    return of({ ...this.tutores[indice] }).pipe(delay(this.atraso));
  }

  remover(id: string): Observable<void> {
    const existe = this.tutores.some((t) => t._id === id);

    if (!existe) {
      return throwError(() => new Error(`Tutor ${id} não encontrado`));
    }

    this.tutores = this.tutores.filter((t) => t._id !== id);

    return of(undefined).pipe(delay(this.atraso));
  }
}

  standalone: true,

  templateUrl: './tutores-formulario.html',
  styleUrl: './tutores-formulario.css',
})
export class TutoresFormulario implements OnInit {
  formulario = new FormGroup({
    nome: new FormControl('', {
      nonNullable: true,
      validators: [
        Validators.required,
        Validators.minLength(3),
        Validators.maxLength(80),
      ],
    }),
    telefone: new FormControl('', {
      nonNullable: true,
      validators: [
        Validators.required,
        Validators.pattern(/^\(\d{2}\) \d{4,5}-\d{4}$/),
      ],
    }),
    email: new FormControl('', {
      nonNullable: true,
      validators: [Validators.required, Validators.email],
    }),
    endereco: new FormControl('', {
      nonNullable: true,
      validators: [Validators.maxLength(150)],
    }),
  });

  idTutor: string | null = null;
  carregando = signal(false);
  salvando = signal(false);
  erro = signal('');

  constructor(
    private tutoresService: TutoresService,
    private router: Router,
    private route: ActivatedRoute,
  ) {}

    if (this.idTutor) {

      next: (tutor) => {
        this.formulario.patchValue({
          nome: tutor.nome,
          telefone: tutor.telefone,
          email: tutor.email,
          endereco: tutor.endereco ?? '',
        });

        this.carregando.set(false);
      },
      error: () => {
        this.erro.set('Não foi possível carregar os dados do tutor.');
        this.carregando.set(false);
      },
    });

      this.formulario.markAllAsTouched();
      return;
    }

    const dados: TutorEntrada = {
      nome: valor.nome.trim(),
      telefone: valor.telefone.trim(),
      email: valor.email.trim(),
      endereco: valor.endereco.trim() || undefined,
    };

    this.salvando.set(true);

    const operacao = this.idTutor
      ? this.tutoresService.atualizar(this.idTutor, dados)
      : this.tutoresService.criar(dados);

      next: () => {

import { Component, OnInit, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { Router, RouterLink } from '@angular/router';

import { Tutor } from '../tutor.model';
import { TutoresService } from '../services/tutores.service';
import { TutorCard } from '../tutor-card/tutor-card';

@Component({
  selector: 'app-tutores-lista',
  standalone: true,
  imports: [CommonModule, FormsModule, RouterLink, TutorCard],
  templateUrl: './tutores-lista.html',
  styleUrl: './tutores-lista.css',
})
export class TutoresLista implements OnInit {
  tutores: Tutor[] = [];
  termoBusca = '';

  carregando = signal(false);
  erro = signal('');
  sucesso = signal('');

  tutorSelecionado: Tutor | null = null;

  constructor(
    private tutoresService: TutoresService,
    private router: Router,
  ) {}

  ngOnInit(): void {
    this.carregarTutores();
  }

  carregarTutores(): void {
    this.carregando.set(true);
    this.erro.set('');
    this.sucesso.set('');

    this.tutoresService.listar().subscribe({
      next: (dados) => {
        this.tutores = dados;
        this.carregando.set(false);
        this.sucesso.set('Lista de tutores carregada com sucesso.');
      },
      error: () => {
        this.erro.set('Não foi possível carregar os tutores.');
        this.carregando.set(false);
      },
    });
  }

  get tutoresFiltrados(): Tutor[] {
    const termo = this.termoBusca.trim().toLocaleLowerCase();

    return this.tutores.filter((tutor) =>
      tutor.nome.toLocaleLowerCase().includes(termo),
    );
  }

  selecionarTutor(tutor: Tutor): void {
    this.tutorSelecionado = tutor;
  }

  limparSelecao(): void {
    this.tutorSelecionado = null;
  }

  editarTutor(tutor: Tutor): void {
    this.router.navigate(['/tutores', tutor._id, 'editar']);
  }

  excluirTutor(tutor: Tutor): void {
    const confirmar = window.confirm(
      `Tem certeza que deseja excluir o tutor ${tutor.nome}?`,
    );

    if (!confirmar) {
      return;
    }

    this.erro.set('');
    this.sucesso.set('');

    this.tutoresService.remover(tutor._id).subscribe({
      next: () => {
        this.tutores = this.tutores.filter(
          (item) => item._id !== tutor._id,
        );

        if (this.tutorSelecionado?._id === tutor._id) {
          this.tutorSelecionado = null;
        }

        this.sucesso.set('Tutor excluído com sucesso.');
      },
      error: () => {
        this.erro.set('Não foi possível excluir o tutor.');
      },
    });
  }
}
